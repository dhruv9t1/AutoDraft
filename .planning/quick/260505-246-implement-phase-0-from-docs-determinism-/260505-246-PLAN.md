   ```rust
   //! Determinism unit test for the document chunker.
   //!
   //! Verifies that `OllamaService::chunk_document` produces byte-identical
   //! output across repeated calls on the same input. This test is part of
   //! Phase 0 (DETERMINISM_AND_ACCURACY_TRACK.md task #5) and runs in CI
   //! because it does not contact Ollama.

   use word_addin_backend::config::Config;
   use word_addin_backend::ollama::OllamaService;

   fn fixture() -> String {
       std::fs::read_to_string("tests/fixtures/determinism_doc.txt")
           .expect("fixture file missing — see backend/tests/fixtures/determinism_doc.txt")
   }

   fn build_service() -> OllamaService {
       // Use defaults — chunker only reads `ollama_chunk_size`.
       // We override env vars to a small chunk size so the chunker actually splits.
       std::env::set_var("OLLAMA_CHUNK_SIZE", "300");
       let cfg = Config::from_env().expect("config from env");
       OllamaService::new(cfg)
   }

   #[test]
   fn chunker_is_deterministic_across_runs() {
       let svc = build_service();
       let doc = fixture();

       let first = svc.chunk_document_for_test(&doc);
       let second = svc.chunk_document_for_test(&doc);
       let third = svc.chunk_document_for_test(&doc);

       assert_eq!(first, second, "chunk_document differed between run 1 and run 2");
       assert_eq!(second, third, "chunk_document differed between run 2 and run 3");
       assert!(!first.is_empty(), "chunker returned no chunks");
   }

   #[test]
   fn chunker_boundaries_stable() {
       let svc = build_service();
       let doc = fixture();
       let chunks = svc.chunk_document_for_test(&doc);

       // Round-trip: concatenating chunks should reconstruct the input
       // (allowing for the documented `\n\n` inclusion behavior — chunks
       // share the trailing newlines on the boundary).
       let rejoined: String = chunks.join("");
       assert_eq!(rejoined.len(), doc.len(),
           "chunk reassembly length mismatch: {} vs {}", rejoined.len(), doc.len());
   }
   ```

   ```rust
   //! Snapshot determinism test for the full LLM pipeline.
   //!
   //! Runs `identify_variables` 10× on a fixed fixture and asserts byte-identical
   //! JSON-serialized variable lists across all 10 runs. This is the Phase 0
   //! exit criterion (DETERMINISM_AND_ACCURACY_TRACK.md line 45).
   //!
   //! NOT run in CI — requires a live Ollama server. To run locally:
   //!
   //!     cd backend
   //!     OLLAMA_DETERMINISM_TEST=1 cargo test --test determinism_snapshot -- --ignored --nocapture
   //!
   //! Both gates must be satisfied:
   //!   1. `#[ignore]` (skipped by default `cargo test`)
   //!   2. `OLLAMA_DETERMINISM_TEST=1` env var (skipped without it even when --ignored)

   use word_addin_backend::config::Config;
   use word_addin_backend::ollama::OllamaService;

   #[tokio::test]
   #[ignore]
   async fn snapshot_10x_byte_identical() {
       if std::env::var("OLLAMA_DETERMINISM_TEST").ok().as_deref() != Some("1") {
           eprintln!("Skipping: set OLLAMA_DETERMINISM_TEST=1 to run this test.");
           return;
       }

       let cfg = Config::from_env().expect("config from env");
       let svc = OllamaService::new(cfg);
       let doc = std::fs::read_to_string("tests/fixtures/determinism_doc.txt")
           .expect("fixture missing");

       let mut snapshots: Vec<String> = Vec::with_capacity(10);
       for i in 0..10 {
           let vars = svc.identify_variables(&doc).await
               .unwrap_or_else(|e| panic!("identify_variables failed on run {}: {:?}", i, e));
           let json = serde_json::to_string(&vars)
               .expect("serialize variables");
           snapshots.push(json);
       }

       let first = &snapshots[0];
       for (i, snap) in snapshots.iter().enumerate().skip(1) {
           assert_eq!(snap, first,
               "Run {} diverged from run 0:\nrun 0: {}\nrun {}: {}",
               i, first, i, snap);
       }
       eprintln!("✅ All 10 runs byte-identical ({} chars each).", first.len());
   }
   ```
